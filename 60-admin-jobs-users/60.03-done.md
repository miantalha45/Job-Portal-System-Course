# Chapter 60 — Definition of Done

## Verifiable checklist

```bash
# 1. TypeScript compiles with zero errors
npx tsc --noEmit

# 2. List all jobs — expect a JSON array
curl -s -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/jobs" | jq '.jobs | length'

# 3. Filter jobs by status
curl -s -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/jobs?status=open" \
  | jq '[.jobs[].status] | unique'
# Expected: ["open"]

# 4. Force-close a job (replace <JOB_ID>)
curl -s -X PATCH -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/jobs/<JOB_ID>/close" \
  | jq '.job.status'
# Expected: "closed"

# 5. Try to close the same job again — expect 409
curl -s -o /dev/null -w "%{http_code}" -X PATCH \
  -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/jobs/<JOB_ID>/close"
# Expected: 409

# 6. List all users
curl -s -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/users" | jq '.users | length'

# 7. Suspend a non-admin user (replace <USER_ID>)
curl -s -X PATCH -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/users/<USER_ID>/suspend" \
  | jq '.user.status'
# Expected: "suspended"

# 8. Confirm refresh tokens deleted in DB
psql $DATABASE_URL -c \
  "SELECT count(*) FROM refresh_tokens WHERE user_id = '<USER_ID>';"
# Expected: count = 0

# 9. Confirm suspended user status in DB
psql $DATABASE_URL -c \
  "SELECT status FROM users WHERE id = '<USER_ID>';"
# Expected: suspended

# 10. Attempt self-suspension — expect 403
ADMIN_ID=$(curl -s -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/auth/me" | jq -r '.user.id')
curl -s -o /dev/null -w "%{http_code}" -X PATCH \
  -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/users/${ADMIN_ID}/suspend"
# Expected: 403

# 11. Reactivate the suspended user
curl -s -X PATCH -H "Authorization: Bearer $TOKEN" \
  "http://localhost:3000/api/admin/users/<USER_ID>/activate" \
  | jq '.user.status'
# Expected: "active"

# 12. Non-admin calling admin route — expect 403
curl -s -o /dev/null -w "%{http_code}" \
  -H "Authorization: Bearer $NON_ADMIN_TOKEN" \
  "http://localhost:3000/api/admin/users"
# Expected: 403
```
