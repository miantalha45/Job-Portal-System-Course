# The seed script

## Why not a random data library?

Tools like Faker generate unlimited realistic-looking data. The temptation is to seed a hundred applicants, fifty jobs, and thousands of applications. The problem: when a query returns wrong data, you can't tell whether the data itself is wrong or the query is broken — because you don't know what the data should look like. Deterministic fixture data — small, specific, written by hand — means you always know the expected output of every query.

## Install bcryptjs

The seed script hashes passwords so the seed users can actually log in once auth is built in [Chapter 19](../19-password-hashing-and-auth/). `bcryptjs` is a pure-JavaScript bcrypt implementation with no native compilation step:

```bash
npm install bcryptjs
npm install --save-dev @types/bcryptjs
```

## Add the npm script

In `package.json`:

```json
{
  "scripts": {
    "seed": "tsx scripts/seed.ts"
  }
}
```

## The script

Create `scripts/seed.ts`. The script imports `config` from the shared config module established in [Chapter 6](../06-config-and-secrets/) — following the rule that only `config.ts` reads `process.env`. It uses a single `Client` (not a `Pool`) because a seed script connects once, does its work, and exits.

```ts
import { Client } from 'pg';
import bcrypt from 'bcryptjs';
import { config } from '../src/shared/config';

const client = new Client({ connectionString: config.DATABASE_URL });

async function seed() {
  await client.connect();

  try {
    await client.query('BEGIN');

    await client.query(`
      TRUNCATE applications, jobs, recruiters, applicants, admins, companies, users
      RESTART IDENTITY CASCADE
    `);

    const hash = await bcrypt.hash('password123', 10);

    // ── Users ─────────────────────────────────────────────────
    // Seed users are set to 'active' to bypass the email-verification
    // flow that real signups go through — covered in Chapter 19.
    const { rows: [adminUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('admin@portal.dev', $1, 'admin', 'active') RETURNING id`,
      [hash],
    );

    const { rows: [aliceUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('alice@brightbuild.dev', $1, 'recruiter', 'active') RETURNING id`,
      [hash],
    );

    const { rows: [carlosUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('carlos@novaspark.dev', $1, 'recruiter', 'active') RETURNING id`,
      [hash],
    );

    const { rows: [jamieUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('jamie@example.dev', $1, 'applicant', 'active') RETURNING id`,
      [hash],
    );

    const { rows: [patUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('pat@example.dev', $1, 'applicant', 'active') RETURNING id`,
      [hash],
    );

    const { rows: [morganUser] } = await client.query<{ id: string }>(
      `INSERT INTO users (email, password_hash, role, status)
       VALUES ('morgan@example.dev', $1, 'applicant', 'active') RETURNING id`,
      [hash],
    );

    // ── Admin ──────────────────────────────────────────────────
    await client.query(
      `INSERT INTO admins (user_id) VALUES ($1)`,
      [adminUser.id],
    );

    // ── Companies ──────────────────────────────────────────────
    const { rows: [brightbuild] } = await client.query<{ id: string }>(`
      INSERT INTO companies (name, slug, website, verified)
      VALUES ('BrightBuild', 'brightbuild', 'https://brightbuild.dev', true)
      RETURNING id
    `);

    const { rows: [novaspark] } = await client.query<{ id: string }>(`
      INSERT INTO companies (name, slug, website, verified)
      VALUES ('NovaSpark', 'novaspark', 'https://novaspark.dev', true)
      RETURNING id
    `);

    // ── Recruiters ─────────────────────────────────────────────
    await client.query(
      `INSERT INTO recruiters (user_id, company_id, company_role)
       VALUES ($1, $2, 'owner')`,
      [aliceUser.id, brightbuild.id],
    );

    await client.query(
      `INSERT INTO recruiters (user_id, company_id, company_role)
       VALUES ($1, $2, 'owner')`,
      [carlosUser.id, novaspark.id],
    );

    // ── Applicants ─────────────────────────────────────────────
    const { rows: [jamie] } = await client.query<{ id: string }>(
      `INSERT INTO applicants (user_id, full_name, headline, location)
       VALUES ($1, 'Jamie Rivera', 'Backend engineer, 4 years experience', 'Remote')
       RETURNING id`,
      [jamieUser.id],
    );

    const { rows: [pat] } = await client.query<{ id: string }>(
      `INSERT INTO applicants (user_id, full_name, headline, location)
       VALUES ($1, 'Pat Chen', 'Full-stack developer', 'New York')
       RETURNING id`,
      [patUser.id],
    );

    const { rows: [morgan] } = await client.query<{ id: string }>(
      `INSERT INTO applicants (user_id, full_name, headline)
       VALUES ($1, 'Morgan Ellis', 'UI/UX designer, product thinker')
       RETURNING id`,
      [morganUser.id],
    );

    // ── Jobs ───────────────────────────────────────────────────
    const { rows: [backendJob] } = await client.query<{ id: string }>(
      `INSERT INTO jobs (company_id, title, description, status)
       VALUES ($1, 'Backend Engineer', 'Build and maintain the core API and data layer.', 'open')
       RETURNING id`,
      [brightbuild.id],
    );

    await client.query(
      `INSERT INTO jobs (company_id, title, description, status)
       VALUES ($1, 'Product Designer', 'Lead design for the web platform.', 'open')`,
      [brightbuild.id],
    );

    const { rows: [uiJob] } = await client.query<{ id: string }>(
      `INSERT INTO jobs (company_id, title, description, status)
       VALUES ($1, 'UI Designer', 'Shape the visual identity of client-facing products.', 'open')
       RETURNING id`,
      [novaspark.id],
    );

    await client.query(
      `INSERT INTO jobs (company_id, title, description, status)
       VALUES ($1, 'Senior Frontend Engineer', 'Own the component library and performance budget.', 'draft')`,
      [novaspark.id],
    );

    // ── Applications ───────────────────────────────────────────
    await client.query(
      `INSERT INTO applications (job_id, applicant_id, stage)
       VALUES ($1, $2, 'screening')`,
      [backendJob.id, jamie.id],
    );

    await client.query(
      `INSERT INTO applications (job_id, applicant_id, stage)
       VALUES ($1, $2, 'interview')`,
      [uiJob.id, jamie.id],
    );

    await client.query(
      `INSERT INTO applications (job_id, applicant_id, stage)
       VALUES ($1, $2, 'applied')`,
      [backendJob.id, pat.id],
    );

    await client.query(
      `INSERT INTO applications (job_id, applicant_id, stage)
       VALUES ($1, $2, 'applied')`,
      [uiJob.id, morgan.id],
    );

    await client.query('COMMIT');
    console.log('Seed complete.');

  } catch (err) {
    await client.query('ROLLBACK').catch(() => {});
    throw err;
  } finally {
    await client.end();
  }
}

seed().catch((err) => {
  console.error(err);
  process.exit(1);
});
```

## What you now have

| Email | Password | Role | Notes |
|---|---|---|---|
| admin@portal.dev | password123 | admin | Platform operator |
| alice@brightbuild.dev | password123 | recruiter | Owner of BrightBuild |
| carlos@novaspark.dev | password123 | recruiter | Owner of NovaSpark |
| jamie@example.dev | password123 | applicant | Applied to two companies, different stages |
| pat@example.dev | password123 | applicant | Applied to Backend Engineer |
| morgan@example.dev | password123 | applicant | Applied to UI Designer |

BrightBuild has 2 jobs (both open). NovaSpark has 2 jobs (1 open, 1 draft). The draft job appears nowhere on the public board — useful for testing the status filter. Jamie has applications at two different stages across two companies — useful for testing both the applicant's own view and the recruiter's inbox in isolation.

## RETURNING and typed results

Every insert that a later insert references uses `RETURNING id` with a type argument:

```ts
const { rows: [brightbuild] } = await client.query<{ id: string }>(`
  INSERT INTO companies (...) VALUES (...) RETURNING id
`);
// brightbuild.id: string — available for recruiter and job inserts
```

Without `<{ id: string }>`, `pg` types `rows` as `any[]` and you lose all type safety on the returned field. Inserts that need no cross-reference — like the Product Designer job — omit `RETURNING` entirely.
