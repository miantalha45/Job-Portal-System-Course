## Full system verifiable checklist

### Phase 1 — Compilation & startup

```bash
# 1. TypeScript compiles across the entire project with zero errors
npx tsc --noEmit
# Expected: no output (zero errors)

# 2. API server starts without errors
npm run dev &

# 3. Worker starts without errors
npm run worker &
WORKER_PID=$!
sleep 3
# Expected: "[worker] Worker started and listening for jobs..."
# Expected: "[worker] Repeatable jobs registered: [...]"

```

### Phase 2 — Auth flow

```bash
# 4. Register a new applicant
curl -s -X POST http://localhost:3000/api/auth/register \
  -H 'Content-Type: application/json' \
  -d '{"email":"alice@example.com","password":"Secret123!","role":"applicant"}' \
  | jq '.message'
# Expected: "Verification email sent"

# 5. Verify email (get the token from email_verifications table)
OTP=$(psql $DATABASE_URL -t -c \
  "SELECT token FROM email_verifications \
   WHERE user_id = (SELECT id FROM users WHERE email='alice@example.com') \
   ORDER BY created_at DESC LIMIT 1;" | tr -d ' \n')

curl -s -X POST http://localhost:3000/api/auth/verify-email \
  -H 'Content-Type: application/json' \
  -d "{\"token\":\"$OTP\"}" | jq '.message'
# Expected: "Email verified"

# 6. Log in and get a token
APPLICANT_TOKEN=$(curl -s -X POST http://localhost:3000/api/auth/login \
  -H 'Content-Type: application/json' \
  -d '{"email":"alice@example.com","password":"Secret123!"}' \
  | jq -r '.accessToken')
echo "Applicant token obtained: ${APPLICANT_TOKEN:0:20}..."
# Expected: a non-null JWT string

# 7. Register a recruiter
curl -s -X POST http://localhost:3000/api/auth/register \
  -H 'Content-Type: application/json' \
  -d '{"email":"bob@example.com","password":"Secret123!","role":"recruiter"}'

OTP2=$(psql $DATABASE_URL -t -c \
  "SELECT token FROM email_verifications \
   WHERE user_id = (SELECT id FROM users WHERE email='bob@example.com') \
   ORDER BY created_at DESC LIMIT 1;" | tr -d ' \n')

curl -s -X POST http://localhost:3000/api/auth/verify-email \
  -H 'Content-Type: application/json' \
  -d "{\"token\":\"$OTP2\"}"

RECRUITER_TOKEN=$(curl -s -X POST http://localhost:3000/api/auth/login \
  -H 'Content-Type: application/json' \
  -d '{"email":"bob@example.com","password":"Secret123!"}' \
  | jq -r '.accessToken')
echo "Recruiter token obtained: ${RECRUITER_TOKEN:0:20}..."
```

### Phase 3 — Company and job creation

```bash
# 8. Create a company as recruiter
COMPANY_ID=$(curl -s -X POST http://localhost:3000/api/companies \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  -d '{"name":"TechCorp"}' | jq -r '.company.id')
echo "Company ID: $COMPANY_ID"
# Expected: a UUID

# 9. Admin verifies the company
psql $DATABASE_URL -c "UPDATE users SET role = 'admin' WHERE email = 'admin@example.com';" 2>/dev/null || true
# (Assuming an admin user exists; otherwise insert one directly)

ADMIN_TOKEN=$(curl -s -X POST http://localhost:3000/api/auth/login \
  -H 'Content-Type: application/json' \
  -d '{"email":"admin@example.com","password":"Secret123!"}' \
  | jq -r '.accessToken')

curl -s -X PATCH \
  -H "Authorization: Bearer $ADMIN_TOKEN" \
  "http://localhost:3000/api/admin/companies/$COMPANY_ID/verify" \
  | jq '.company.status'
# Expected: "verified"

# 10. Post a job
JOB_ID=$(curl -s -X POST http://localhost:3000/api/companies/jobs \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  -d '{"title":"Senior Backend Engineer","description":"Build APIs","status":"open"}' \
  | jq -r '.job.id')
echo "Job ID: $JOB_ID"
# Expected: a UUID

# 11. Job appears on the public job board
curl -s "http://localhost:3000/api/jobs" | jq '[.jobs[] | select(.id == "'$JOB_ID'")] | length'
# Expected: 1
```

### Phase 4 — Applicant experience

```bash
# 12. Create an applicant profile
curl -s -X POST http://localhost:3000/api/applicants/profile \
  -H "Authorization: Bearer $APPLICANT_TOKEN" \
  -H 'Content-Type: application/json' \
  -d '{"headline":"Node.js engineer","bio":"5 years exp","location":"Remote"}'
# Expected: 201 with profile data

# 13. Apply to the job
APP_ID=$(curl -s -X POST http://localhost:3000/api/applications \
  -H "Authorization: Bearer $APPLICANT_TOKEN" \
  -H 'Content-Type: application/json' \
  -d "{\"jobId\":\"$JOB_ID\",\"answers\":{}}" | jq -r '.application.id')
echo "Application ID: $APP_ID"
# Expected: a UUID

# 14. Application is idempotent (second apply returns same ID, not duplicate)
SECOND_APP_ID=$(curl -s -X POST http://localhost:3000/api/applications \
  -H "Authorization: Bearer $APPLICANT_TOKEN" \
  -H 'Content-Type: application/json' \
  -d "{\"jobId\":\"$JOB_ID\",\"answers\":{}}" | jq -r '.application.id')
[ "$APP_ID" = "$SECOND_APP_ID" ] && echo "IDEMPOTENT OK" || echo "IDEMPOTENT FAIL"
# Expected: "IDEMPOTENT OK"

# 15. Application is in DB with snapshot
psql $DATABASE_URL -c \
  "SELECT id, stage, status, snapshot->>'title' FROM applications WHERE id = '$APP_ID';"
# Expected: row with stage='applied', status='active', snapshot title present
```

### Phase 5 — Stage pipeline and interview

```bash
# 16. Recruiter moves application to 'screening'
curl -s -X PATCH \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  "http://localhost:3000/api/applications/$APP_ID/stage" \
  -d '{"stage":"screening"}' | jq '.application.stage'
# Expected: "screening"

# 17. Move to 'interview'
curl -s -X PATCH \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  "http://localhost:3000/api/applications/$APP_ID/stage" \
  -d '{"stage":"interview"}' | jq '.application.stage'
# Expected: "interview"

# 18. Schedule an interview (worker must be running)
INTERVIEW_ID=$(curl -s -X POST http://localhost:3000/api/interviews \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  -d "{
    \"applicationId\":\"$APP_ID\",
    \"scheduledAt\":\"2026-09-01T10:00:00.000Z\",
    \"meetingLink\":\"https://meet.example.com/final\"
  }" | jq -r '.interview.id')
echo "Interview ID: $INTERVIEW_ID"
# Expected: a UUID

# 19. Interview notification job was queued and processed
sleep 3
npx ts-node -e "
import queue from './src/shared/queue';
(async () => {
  const completed = await queue.getCompleted(0, 20);
  const found = completed.find(j => j.name === 'send-interview-notification');
  console.log('Interview notification sent:', !!found);
  await queue.close();
})();
"
# Expected: "Interview notification sent: true"

# 20. Submit interview feedback
curl -s -X POST \
  -H "Authorization: Bearer $RECRUITER_TOKEN" \
  -H 'Content-Type: application/json' \
  "http://localhost:3000/api/interviews/$INTERVIEW_ID/feedback" \
  -d '{"outcome":"passed","notes":"Strong technical skills"}' \
  | jq '.interview.outcome'
# Expected: "passed"
```

```bash

for i in {1..11}; do
  CODE=$(curl -s -o /dev/null -w "%{http_code}" \
    -X POST http://localhost:3000/api/auth/login \
    -H 'Content-Type: application/json' \
    -d '{"email":"nobody@example.com","password":"wrong"}')
  echo "Request $i: $CODE"
done
# Expected: requests 1-10 return 401, request 11 returns 429

# 28. Docker build succeeds for both images
docker build -t jobportal-api:final -f Dockerfile .
docker build -t jobportal-worker:final -f Dockerfile.worker .
# Expected: both build successfully

# 29. No direct process.env access in any module (except config.ts)
grep -rn 'process\.env' src/modules/ src/shared/ | grep -v 'config\.ts'
# Expected: no output

# 30. Worker graceful shutdown on SIGTERM
npm run worker &
WORKER_PID=$!
sleep 3
kill -SIGTERM $WORKER_PID
wait $WORKER_PID
echo "Worker exit code: $?"
# Expected: "Worker closed" log line, exit code 0
```
