# Chapter 71 — Structured Logging & Request IDs

## What you will build

Upgrade the logger to `pino`, add a request ID middleware that stamps every log line with a
unique request ID, and define a consistent log format for request lifecycle events and errors.

## Learning goals

- Replace `console.log` with structured JSON logging
- Generate a unique ID per request and propagate it through all log calls for that request
- Understand the difference between development-friendly pretty-print and production JSON

## Prerequisites

- An existing `src/shared/logger.ts` (even if it wraps `console.log`)
- Chapter 6: `config.ts` — add `LOG_LEVEL`

## Implementation

### src/shared/logger.ts

```ts
import pino from 'pino';
import config from './config';

const logger = pino({
  level: config.LOG_LEVEL,
  transport:
    config.NODE_ENV !== 'production'
      ? { target: 'pino-pretty', options: { colorize: true } }
      : undefined,
});

export default logger;
```

`pino-pretty` is only used in non-production environments. In production, pino emits
newline-delimited JSON — one JSON object per log line. Log aggregators (Datadog, CloudWatch,
Loki) parse this automatically.

### src/middleware/requestId.ts

```ts
import { Request, Response, NextFunction } from 'express';

declare global {
  namespace Express {
    interface Request {
      id: string;
    }
  }
}

// Attach a unique request ID to every incoming request
export function requestIdMiddleware(req: Request, _res: Response, next: NextFunction) {
  req.id = crypto.randomUUID();
  next();
}
```

`crypto.randomUUID()` is available in Node.js 14.17+ with no import.

### src/middleware/httpLogger.ts

```ts
import { Request, Response, NextFunction } from 'express';
import logger from '../shared/logger';

// Log request received and response completed with timing
export function httpLogger(req: Request, res: Response, next: NextFunction) {
  const start = Date.now();

  logger.info({ requestId: req.id, method: req.method, url: req.url }, 'request received');

  res.on('finish', () => {
    const durationMs = Date.now() - start;
    logger.info(
      {
        requestId: req.id,
        method: req.method,
        url: req.url,
        statusCode: res.statusCode,
        durationMs,
      },
      'request completed'
    );
  });

  next();
}
```

### app.ts — apply middleware in order

```ts
import { requestIdMiddleware } from './middleware/requestId';
import { httpLogger } from './middleware/httpLogger';

app.use(requestIdMiddleware);   // first: assign the ID
app.use(httpLogger);            // second: log with the ID
app.use(globalLimiter);         // rate limiter
// ... routes
```

### Global error handler — log with request ID

In the global error handler (`src/middleware/errorHandler.ts`):

```ts
import logger from '../shared/logger';

export function errorHandler(err: Error, req: Request, res: Response, _next: NextFunction) {
  logger.error({ requestId: req.id, err }, 'unhandled error');
  // ... existing error response logic
}
```

## Log format (production JSON)

Request received:
```json
{ "level": 30, "time": 1749283200000, "requestId": "550e8400-...", "method": "POST", "url": "/api/auth/login", "msg": "request received" }
```

Request completed:
```json
{ "level": 30, "time": 1749283200050, "requestId": "550e8400-...", "method": "POST", "url": "/api/auth/login", "statusCode": 200, "durationMs": 48, "msg": "request completed" }
```

Unhandled error:
```json
{ "level": 50, "time": 1749283200051, "requestId": "550e8400-...", "err": { "type": "Error", "message": "...", "stack": "..." }, "msg": "unhandled error" }
```

## Replacing console.log

Search the codebase for any remaining `console.log` calls in `src/modules/` or `src/shared/`:

```bash
grep -rn 'console\.log\|console\.error' src/modules/ src/shared/
```

Replace each with the appropriate `logger.info(...)` or `logger.error(...)` call. Worker
`console.log` calls may be replaced with a worker-specific logger instance.
